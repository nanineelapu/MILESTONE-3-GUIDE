pipeline {
    agent any
    tools {
        maven 'Maven'
    }
    stages {
        stage('Stage 1: Clone Source Code') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/Msocial123/EcommerceApp.git'
            }
        }
        stage('Stage 2: SonarQube Analysis') {
            steps {
                withSonarQubeEnv('SonarQube') {
                    sh 'mvn sonar:sonar'
                }
            }
        }
        stage('Stage 2.1: Quality Gate') {
            steps {
                timeout(time: 2, unit: 'MINUTES') {
                    waitForQualityGate abortPipeline: true
                }
            }
        }
        stage('Stage 3: Maven Build') {
            steps {
                sh 'mvn clean package -DskipTests'
            }
        }
        stage('Stage 4: Ansible Deploy') {
            steps {
                sh 'ansible-playbook /etc/ansible/deploy.yml'
            }
        }
    }
    post {
        success { echo 'Deployment Successful!' }
        failure { echo 'Pipeline Failed!' }
    }
}


---
- hosts: appserver
  become: yes
  tasks:
    - name: Stop Tomcat
      service:
        name: tomcat
        state: stopped
    - name: Copy WAR to Tomcat
      copy:
        src: /var/lib/jenkins/workspace/EcommerceApp/target/EcommerceApp.war
        dest: /var/lib/tomcat/webapps/
        remote_src: no
    - name: Start Tomcat
      service:
        name: tomcat
        state: started


[appserver]
<VM3-Public-IP> ansible_user=ec2-user ansible_ssh_private_key_file=/var/lib/jenkins/.ssh/id_rsa
