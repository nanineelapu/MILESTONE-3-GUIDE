✅ PHASE 5 — Create Ansible Playbook on VM1 (10 mins)
Still on VM1:
bash# Create inventory file
sudo nano /etc/ansible/hosts
Add this:
ini[appserver]
<VM3-Public-IP> ansible_user=ec2-user ansible_ssh_private_key_file=/var/lib/jenkins/.ssh/id_rsa
bash# Create deploy playbook
sudo nano /etc/ansible/deploy.yml
Paste this:
yaml---
- hosts: appserver
  become: yes
  tasks:
    - name: Stop Tomcat
      service:
        name: tomcat
        state: stopped

    - name: Copy WAR to Tomcat
      copy:
        src: /var/lib/jenkins/workspace/EcommerceApp/target/EcommerceApp.war
        dest: /var/lib/tomcat/webapps/
        remote_src: no

    - name: Start Tomcat
      service:
        name: tomcat
        state: started

    - name: Verify Tomcat is running
      service:
        name: tomcat
        state: started

✅ PHASE 7 — Create Jenkins Pipeline (10 mins)

Jenkins dashboard → New Item
Name: EcommerceApp
Select Pipeline → OK

Scroll down to Pipeline script and paste:
groovypipeline {
    agent any

    tools {
        maven 'Maven'
    }

    stages {

        stage('Stage 1: Clone Source Code') {
            steps {
                echo 'Cloning repository...'
                git branch: 'main',
                    url: 'https://github.com/Msocial123/EcommerceApp.git'
            }
        }

        stage('Stage 2: SonarQube Code Quality') {
            steps {
                echo 'Running SonarQube analysis...'
                withSonarQubeEnv('SonarQube') {
                    sh 'mvn sonar:sonar'
                }
            }
        }

        stage('Stage 2.1: Quality Gate') {
            steps {
                echo 'Checking Quality Gate...'
                timeout(time: 2, unit: 'MINUTES') {
                    waitForQualityGate abortPipeline: true
                }
            }
        }

        stage('Stage 3: Maven Build') {
            steps {
                echo 'Building application...'
                sh 'mvn clean package -DskipTests'
            }
        }

        stage('Stage 4: Ansible Deploy') {
            steps {
                echo 'Deploying via Ansible...'
                sh 'ansible-playbook /etc/ansible/deploy.yml'
            }
        }
    }

    post {
        success {
            echo '✅ Pipeline Successful - App Deployed!'
        }
        failure {
            echo '❌ Pipeline Failed - Check logs'
        }
    }
}
Click Save
